import { useState } from 'react'

export const districts = [
  { name: '旧灯塔', kind: '观察', time: 20, color: '#9a4b23', description: '登上潮湿的旋梯，辨认雾里第三次出现的灯光。' },
  { name: '玻璃集市', kind: '交涉', time: 35, color: '#32665b', description: '用一段真实的记忆，交换一封尚未寄出的信。' },
  { name: '沉钟码头', kind: '探索', time: 45, color: '#515e9a', description: '潮水退去之前，寻找水面下那座没有门的钟楼。' },
]

export function Note({ title = '航海者批注', children }) {
  return <aside className="fog-note"><strong>{title}</strong><div>{children}</div></aside>
}

export function DistrictCards() {
  return (
    <div className="fog-grid">
      {districts.map((district, index) => (
        <article className="fog-card" key={district.name} style={{ borderTopColor: district.color }}>
          <span className="fog-eyebrow">0{index + 1} / {district.kind}</span>
          <h3>{district.name}</h3>
          <p>{district.description}</p>
          <small>步行约 {district.time} 分钟</small>
        </article>
      ))}
    </div>
  )
}

export function DepartureDesk() {
  const [name, setName] = useState('旅人')
  const [route, setRoute] = useState(0)
  const [courage, setCourage] = useState(60)
  const [lantern, setLantern] = useState(true)
  const [ticket, setTicket] = useState(null)
  const destination = districts[route]
  const reset = () => {
    setName('旅人')
    setRoute(0)
    setCourage(60)
    setLantern(true)
    setTicket(null)
  }
  return (
    <section className="fog-desk" aria-label="交互式出航登记">
      <div className="fog-grid fog-form">
        <label>旅人称呼<input value={name} maxLength={24} onChange={event => setName(event.target.value)} placeholder="留下一个名字" /></label>
        <label>目的地<select value={route} onChange={event => setRoute(Number(event.target.value))}>{districts.map((district, index) => <option key={district.name} value={index}>{district.name}</option>)}</select></label>
      </div>
      <label className="fog-range">勇气储备：{courage}%<input type="range" min="0" max="100" step="5" value={courage} onChange={event => setCourage(Number(event.target.value))} /></label>
      <label className="fog-check"><input type="checkbox" checked={lantern} onChange={event => setLantern(event.target.checked)} />带上风灯</label>
      <div className="fog-preview" aria-live="polite">
        <strong>{name.trim() || '无名旅人'}，今晚前往{destination.name}。</strong>
        <p>{courage >= 50 ? '你已准备好主动敲响那扇门。' : '不必勉强，沿着有灯的街道慢慢走。'}{lantern ? '风灯会照亮脚下的石阶。' : '没有风灯，记得在天黑前回来。'}</p>
        <small>预计步行 {destination.time} 分钟 · {destination.kind}路线</small>
      </div>
      <div className="fog-actions">
        <button type="button" disabled={!name.trim()} onClick={() => setTicket({ name: name.trim(), destination: destination.name, courage, lantern })}>生成船票</button>
        <button type="button" className="fog-secondary" onClick={reset}>重置演示</button>
      </div>
      <div role="status">
        {ticket ? <p className="fog-ticket">船票已签发：{ticket.name} → {ticket.destination}，勇气 {ticket.courage}%，{ticket.lantern ? '携带风灯' : '未带风灯'}。这是点击时的快照，修改选项后可重新签发。</p> : <p className="fog-muted">尚未签发船票。填写称呼后即可生成；所有状态仅保留在当前页面。</p>}
      </div>
    </section>
  )
}

<h2 id="prose">01 / 文字有自己的节奏</h2>

## 一座城市的三个细节

### 先听见，再看见

远处的汽笛响了三声。窗台上的 <mark>蓝色盐粒</mark> 开始发光，店主把门牌从“营业”翻成了“等你”。她说：<q>在雾散之前，你可以问一个问题。</q>

- **声音**：钟声比潮汐早到七分钟。
- **气味**：烤面包、海盐，以及刚刚熄灭的火柴。
- **触感**：信封边缘粗糙，封蜡却像刚被人握过。

### 到港后的第一件事

1. 在旅馆留下一个称呼。
2. 找到愿意回答问题的人。
3. 决定是否打开那封信。

这里可以混排 `行内代码`、**粗体**、*斜体*、<del>已经作废的约定</del>，以及 H<sub>2</sub>O、m<sup>2</sup> 等语义标记。键盘标记也可以写成 <kbd>Tab</kbd>，用于提示如何移动焦点。
